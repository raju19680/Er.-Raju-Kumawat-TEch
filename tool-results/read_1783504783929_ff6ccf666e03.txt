   187→function formatDate(dateStr: string): string {
   188→  try {
   189→    return new Date(dateStr).toLocaleDateString('en-IN', {
   190→      month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit',
   191→    })
   192→  } catch { return dateStr }
   193→}
   194→
   195→function formatFileSize(bytes: number | null): string {
   196→  if (!bytes) return '—'
   197→  if (bytes < 1024) return `${bytes} B`
   198→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   199→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   200→}

   212→function getStatusBadge(status: string) {
   213→  switch (status) {
   214→    case 'active':
   215→    case 'completed':
   216→      return <Badge className="bg-emerald-500/15 text-emerald-600 hover:bg-emerald-500/15 border-0 text-[11px]">{status === 'completed' ? 'Completed' : 'Active'}</Badge>
   217→    case 'pending':
   218→      return <Badge className="bg-amber-500/15 text-amber-600 hover:bg-amber-500/15 border-0 text-[11px]">Pending</Badge>
   219→    case 'verifying':
   220→    case 'generating':
   221→      return <Badge className="bg-sky-500/15 text-sky-600 hover:bg-sky-500/15 border-0 text-[11px]">{status === 'verifying' ? 'Verifying' : 'Generating'}</Badge>
   222→    case 'failed':
   223→    case 'error':
   224→      return <Badge className="bg-red-500/15 text-red-600 hover:bg-red-500/15 border-0 text-[11px]">{status === 'failed' ? 'Failed' : 'Error'}</Badge>
   225→    default:
   226→      return <Badge variant="secondary" className="text-[11px]">{status}</Badge>
   227→  }
   228→}

    74→// ─── Password Strength Indicator ────────────────────────────────────────────
    75→function PasswordStrengthIndicator({ password }: { password: string }) {
    76→  const strength = useMemo(() => {
    77→    if (!password) return { score: 0, feedback: [] as string[] }
    78→    const feedback: string[] = []
    79→    let metCount = 0
    80→    if (password.length >= 8) metCount++; else feedback.push('8+ characters')
    81→    if (/[A-Z]/.test(password)) metCount++; else feedback.push('uppercase letter')
    82→    if (/[a-z]/.test(password)) metCount++; else feedback.push('lowercase letter')
    83→    if (/\d/.test(password)) metCount++; else feedback.push('number')
    84→    if (/[^A-Za-z0-9]/.test(password)) metCount++; else feedback.push('special character')
    85→    let score: number
    86→    if (metCount <= 2) { score = 1 }
    87→    else if (metCount === 3) { score = 2 }
    88→    else if (metCount === 4) { score = 3 }
    89→    else { score = 4 }
    90→    return { score, feedback }
    91→  }, [password])
    92→
    93→  if (!password) return null
    94→  const colors = ['bg-red-500', 'bg-orange-500', 'bg-yellow-500', 'bg-emerald-400', 'bg-emerald-600']
    95→  const labels = ['Weak', 'Fair', 'Good', 'Strong', 'Very Strong']
    96→  const textColors = ['text-red-600', 'text-orange-600', 'text-yellow-600', 'text-emerald-500', 'text-emerald-700']
    97→
    98→  return (
    99→    <div className="space-y-1.5">
   100→      <div className="flex gap-1">
   101→        {[0, 1, 2, 3, 4].map((i) => (
   102→          <div key={i} className={`h-1.5 flex-1 rounded-full transition-all duration-300 ${i <= strength.score ? colors[strength.score] : 'bg-gray-200'}`} />
   103→        ))}
   104→      </div>
   105→      <div className="flex items-center justify-between">
   106→        <span className={`text-xs font-medium ${textColors[strength.score]}`}>{labels[strength.score]}</span>
   107→        {strength.feedback.length > 0 && strength.score < 4 && (
   108→          <span className="text-xs text-gray-400">Need: {strength.feedback.join(', ')}</span>
   109→        )}
   110→      </div>
   111→    </div>
   112→  )
   113→}

   196→function StudentAuthDialog({ open, onOpenChange, initialView, orgData }: {
   197→  open: boolean
   198→  onOpenChange: (open: boolean) => void
   199→  initialView?: AuthView
   200→  orgData: OrgData
   201→}) {
   202→  const loginToStore = useAppStore((s) => s.login)
   203→  const [authView, setAuthView] = useState<AuthView>(initialView || 'login')
   204→
   205→  // Login form state
   206→  const [orgId, setOrgId] = useState('')
   207→  const [email, setEmail] = useState('')
   208→  const [password, setPassword] = useState('')
   209→  const [showPassword, setShowPassword] = useState(false)
   210→  const [authLoading, setAuthLoading] = useState(false)
   211→  const [authError, setAuthError] = useState('')
   212→  const [verifiedOrg, setVerifiedOrg] = useState<OrgData | null>(null)
   213→  const [verifyingOrg, setVerifyingOrg] = useState(false)
   214→
   215→  // Forgot password state
   216→  const [fpOrgId, setFpOrgId] = useState('')
   217→  const [fpEmail, setFpEmail] = useState('')
   218→  const [fpLoading, setFpLoading] = useState(false)
   219→  const [fpError, setFpError] = useState('')
   220→  const [fpSuccess, setFpSuccess] = useState(false)
   221→  const [fpVerifiedOrg, setFpVerifiedOrg] = useState<OrgData | null>(null)
   222→  const [fpVerifyingOrg, setFpVerifyingOrg] = useState(false)
   223→
   224→  // Reset password state
   225→  const [resetToken, setResetToken] = useState('')
   226→  const [resetUser, setResetUser] = useState<{ name: string; email: string } | null>(null)
   227→  const [tokenValidating, setTokenValidating] = useState(false)
   228→  const [tokenError, setTokenError] = useState('')
   229→  const [rpPassword, setRpPassword] = useState('')
   230→  const [rpConfirmPassword, setRpConfirmPassword] = useState('')
   231→  const [rpShowPassword, setRpShowPassword] = useState(false)
   232→  const [rpShowConfirm, setRpShowConfirm] = useState(false)
   233→  const [rpLoading, setRpLoading] = useState(false)
   234→  const [rpError, setRpError] = useState('')
   235→
   236→  // Sync with initialView prop changes
   237→  useEffect(() => {
   238→    if (initialView) setAuthView(initialView)
   239→  }, [initialView])
   240→
   241→  // Reset state when dialog closes
   242→  const handleOpenChange = useCallback((newOpen: boolean) => {
   243→    if (!newOpen) {
   244→      setAuthView('login')
   245→      setOrgId('')
   246→      setEmail('')
   247→      setPassword('')
   248→      setAuthError('')
   249→      setVerifiedOrg(null)
   250→      setFpOrgId('')
   251→      setFpEmail('')
   252→      setFpError('')
   253→      setFpSuccess(false)
   254→      setFpVerifiedOrg(null)
   255→      setResetToken('')
   256→      setResetUser(null)
   257→      setTokenError('')
   258→      setRpPassword('')
   259→      setRpConfirmPassword('')
   260→      setRpError('')
   261→    }
   262→    onOpenChange(newOpen)
   263→  }, [onOpenChange])
   264→
   265→  // Verify org when orgId changes (debounced) - login form
   266→  useEffect(() => {
   267→    const trimmed = orgId.trim()
   268→    if (!trimmed) { setVerifiedOrg(null); return }
   269→    const timer = setTimeout(async () => {
   270→      setVerifyingOrg(true)
   271→      try {
   272→        const res = await fetch('/api/auth/verify-org', {
   273→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   274→          body: JSON.stringify({ orgCode: trimmed }),
   275→        })
   276→        const data = await res.json()
   277→        setVerifiedOrg(data.success ? data.organization : null)
   278→      } catch { setVerifiedOrg(null) }
   279→      finally { setVerifyingOrg(false) }
   280→    }, 500)
   281→    return () => clearTimeout(timer)
   282→  }, [orgId])
   283→
   284→  // Verify org for forgot password form (debounced)
   285→  useEffect(() => {
   286→    const trimmed = fpOrgId.trim()
   287→    if (!trimmed) { setFpVerifiedOrg(null); return }
   288→    const timer = setTimeout(async () => {
   289→      setFpVerifyingOrg(true)
   290→      try {
   291→        const res = await fetch('/api/auth/verify-org', {
   292→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   293→          body: JSON.stringify({ orgCode: trimmed }),
   294→        })
   295→        const data = await res.json()
   296→        setFpVerifiedOrg(data.success ? data.organization : null)
   297→      } catch { setFpVerifiedOrg(null) }
   298→      finally { setFpVerifyingOrg(false) }
   299→    }, 500)
   300→    return () => clearTimeout(timer)
   301→  }, [fpOrgId])
   302→
   303→
   304→
   305→  // Handle Login
   306→  const handleLogin = async () => {
   307→    if (!orgId.trim()) { setAuthError('Please enter your Institute ID'); return }
   308→    if (!email.trim()) { setAuthError('Please enter your email'); return }
   309→    if (!password.trim()) { setAuthError('Please enter your password'); return }
   310→
   311→    setAuthLoading(true)
   312→    setAuthError('')
   313→    try {
   314→      const res = await fetch('/api/auth/direct-login', {
   315→        method: 'POST',
   316→        headers: { 'Content-Type': 'application/json' },
   317→        body: JSON.stringify({
   318→          email: email.trim().toLowerCase(),
   319→          password: password,
   320→          orgId: orgId.trim(),
   321→        }),
   322→      })
   323→      const data = await res.json()
   324→
   325→      if (!data.success) {
   326→        setAuthError(data.message || 'Invalid email or password')
   327→        setAuthLoading(false)
   328→        return
   329→      }
   330→
   331→      // Login succeeded - use returned user data directly
   332→      const u = data.user
   333→      loginToStore(
   334→        u.orgCode || '',
   335→        u.orgName || 'Er. Raju Kumawat Tech',
   336→        u.name || email.trim().split('@')[0],
   337→        u.email || email.trim().toLowerCase(),
   338→        u.role || 'student',
   339→        u.loginMode || 'student'
   340→      )
   341→
   342→      // Fetch API token (non-critical)
   343→      try {
   344→        const tokenRes = await fetch('/api/auth/session-token', {
   345→          method: 'POST',
   346→          headers: { 'Content-Type': 'application/json' },
   347→          body: JSON.stringify({ email: u.email, orgId: u.orgId }),
   348→        })
   349→        const tokenData = await tokenRes.json()
   350→        if (tokenData.success && tokenData.token) {
   351→          useAppStore.getState().setApiToken(tokenData.token)
   352→        }
   353→      } catch {}
   354→
   355→      setAuthLoading(false)
   356→      onOpenChange(false)
   357→    } catch {
   358→      setAuthError('Something went wrong. Please try again.')
   359→      setAuthLoading(false)
   360→    }
   361→  }
   362→
   363→  // Handle Forgot Password
   364→  const handleForgotPassword = async () => {
   365→    if (!fpOrgId.trim()) { setFpError('Please enter your Institute ID'); return }
   366→    if (!fpEmail.trim()) { setFpError('Please enter your email address'); return }
   367→
   368→    setFpLoading(true)
   369→    setFpError('')
   370→    try {
   371→      const res = await fetch('/api/auth/forgot-password', {
   372→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   373→        body: JSON.stringify({ email: fpEmail.trim().toLowerCase(), orgId: fpOrgId.trim(), source: 'student' }),
   374→      })
   375→      const data = await res.json()
   376→      if (data.success) { setFpSuccess(true) }
   377→      else { setFpError(data.error || 'Something went wrong. Please try again.') }
   378→    } catch {
   379→      setFpError('Network error. Please check your connection and try again.')
   380→    } finally { setFpLoading(false) }
   381→  }
   382→
   383→  // Handle Reset Password
   384→  const handleResetPassword = async () => {
   385→    if (!rpPassword) { setRpError('Please enter a new password'); return }
   386→    if (rpPassword.length < 8) { setRpError('Password must be at least 8 characters long'); return }
   387→    if (rpPassword !== rpConfirmPassword) { setRpError('Passwords do not match'); return }
   388→
   389→    setRpLoading(true)
   390→    setRpError('')
   391→    try {
   392→      const res = await fetch('/api/auth/reset-password', {
   393→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   394→        body: JSON.stringify({ token: resetToken, password: rpPassword }),
   395→      })
   396→      const data = await res.json()
   397→      if (data.success) { setAuthView('reset-success') }
   398→      else { setRpError(data.error || 'Failed to reset password. Please try again.') }
   399→    } catch {
   400→      setRpError('Network error. Please check your connection and try again.')
   401→    } finally { setRpLoading(false) }
   402→  }
   403→
   404→  // Navigate views
   405→  const goToForgotPassword = () => {
   406→    setAuthView('forgot-password')
   407→    if (orgId.trim()) setFpOrgId(orgId.trim())
   408→    if (email.trim()) setFpEmail(email.trim())
   409→    setFpError('')
   410→    setFpSuccess(false)
   411→  }
   412→  const goToLogin = () => { setAuthView('login'); setAuthError('') }
   413→
   414→  // Load reset token from URL

   416→    if (!open) return

   421→      setAuthView('reset-password')

   423→      fetch(`/api/auth/reset-password?token=${encodeURIComponent(token)}`)
   424→        .then(res => res.json())
   425→        .then(data => {
   426→          if (data.success && data.user) { setResetUser(data.user); setTokenError('') }
   427→          else { setTokenError(data.error || 'This reset link is invalid or has expired.') }
   428→        })
   429→        .catch(() => { setTokenError('Failed to validate reset link. Please try again.') })
   430→        .finally(() => { setTokenValidating(false) })
   431→      // Clean URL
   432→      window.history.replaceState({}, '', '/')
   433→    }
   434→  }, [open])
   435→
   436→  return (
   437→    <Dialog open={open} onOpenChange={handleOpenChange}>
   438→      <DialogContent className="sm:max-w-md p-0 overflow-hidden">
   439→        {/* Header banner */}
   440→        <div className="bg-gradient-to-r from-blue-600 to-indigo-600 px-6 py-4">
   441→          <div className="flex items-center gap-3">
   442→            <div className="flex items-center justify-center w-10 h-10 rounded-lg bg-white/20">
   443→              <GraduationCap className="size-5 text-white" />

   445→            <div>
   446→              <DialogTitle className="text-white font-bold">
   447→                {authView === 'login' && 'Sign In'}
   448→                {authView === 'forgot-password' && 'Forgot Password'}
   449→                {authView === 'reset-password' && 'Reset Password'}
   450→                {authView === 'reset-success' && 'Success!'}
   451→              </DialogTitle>
   452→              <DialogDescription className="text-blue-100 text-xs">
   453→                {authView === 'login' && 'Access your student portal'}
   454→                {authView === 'forgot-password' && 'Reset your password via email'}
   455→                {authView === 'reset-password' && 'Create a new password'}
   456→                {authView === 'reset-success' && 'Your password has been reset'}
   457→              </DialogDescription>
   458→            </div>
   459→          </div>
   460→        </div>
   461→
   462→        <div className="p-6 max-h-[70vh] overflow-y-auto">
   463→          <AnimatePresence mode="wait">
   464→            {/* ── Login View ── */}
   465→            {authView === 'login' && (
   466→              <motion.div key="login" initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }} className="space-y-4">
   467→                <form onSubmit={(e) => { e.preventDefault(); handleLogin() }} className="space-y-4">
   468→                  <div className="space-y-2">
   469→                    <Label className="text-sm font-medium text-gray-700">Institute ID <span className="text-red-500">*</span></Label>
   470→                    <div className="relative">
   471→                      <KeyRound className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-gray-400" />
   472→                      <Input value={orgId} onChange={(e) => { setOrgId(e.target.value); if (authError) setAuthError('') }}
   473→                        placeholder="Enter institute code" className="h-11 pl-10 pr-10 rounded-lg" disabled={authLoading} required />
   474→                      {verifyingOrg && <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 size-4 text-gray-400 animate-spin" />}
   475→                    </div>
   476→                    <AnimatePresence>{verifiedOrg && <VerifiedOrgBadge org={verifiedOrg} />}</AnimatePresence>
   477→                  </div>
   478→                  <div className="space-y-2">
   479→                    <Label className="text-sm font-medium text-gray-700">Email <span className="text-red-500">*</span></Label>
   480→                    <Input type="email" value={email} onChange={(e) => { setEmail(e.target.value); if (authError) setAuthError('') }}

  1293→// ─── Main Component ──────────────────────────────────────────────────────────
  1294→export default function PublicPortal() {
  1295→  const [data, setData] = useState<PortalData | null>(null)
  1296→  const [loading, setLoading] = useState(true)
  1297→  const [error, setError] = useState(false)
  1298→  const [activePage, setActivePage] = useState<PortalPage>('home')
  1299→  const [authDialogOpen, setAuthDialogOpen] = useState(false)
  1300→  const [authInitialView, setAuthInitialView] = useState<AuthView>('login')
  1301→
  1302→  const openAuthDialog = useCallback((view?: AuthView) => {
  1303→    setAuthInitialView(view || 'login')
  1304→    setAuthDialogOpen(true)
  1305→  }, [])
  1306→
  1307→  const fetchData = useCallback(async () => {
  1308→    setLoading(true)
  1309→    setError(false)
  1310→    try {
  1311→      const res = await fetch('/api/student/portal-data?orgCode=9680177120')
  1312→      const json = await res.json()
  1313→      if (json.success && json.data) {
  1314→        setData(json.data)
  1315→      } else {
  1316→        setError(true)
  1317→      }
  1318→    } catch {
  1319→      setError(true)
  1320→    } finally {
  1321→      setLoading(false)
  1322→    }
  1323→  }, [])
  1324→
  1325→  useEffect(() => { fetchData() }, [fetchData])
  1326→
  1327→  // Check for reset_token in URL on mount — auto-open auth dialog with reset view
  1328→  useEffect(() => {
  1329→    if (typeof window === 'undefined') return
  1330→    const params = new URLSearchParams(window.location.search)
  1331→    const token = params.get('reset_token')
  1332→    if (token) {
  1333→      openAuthDialog('reset-password')
  1334→    }
  1335→  }, [openAuthDialog])
  1336→
  1337→  // Check for redirect from main login after student password reset
  1338→  useEffect(() => {
  1339→    if (typeof window === 'undefined') return
  1340→    const openStudentLogin = sessionStorage.getItem('erkt_open_student_login')
  1341→    if (openStudentLogin === 'true') {
  1342→      sessionStorage.removeItem('erkt_open_student_login')
  1343→      // Small delay to ensure the component is fully mounted
  1344→      setTimeout(() => openAuthDialog('login'), 100)
  1345→    }
  1346→  }, [openAuthDialog])
  1347→
  1348→  // Scroll to top on page change
  1349→  useEffect(() => { window.scrollTo({ top: 0, behavior: 'smooth' }) }, [activePage])
  1350→
  1351→  if (loading) {
  1352→    return (
  1353→      <div className="min-h-screen flex flex-col bg-white">
  1354→        <NavbarSkeleton />
  1355→        <div className="flex-1 px-4 sm:px-6 lg:px-8 py-8">
  1356→          <div className="mx-auto max-w-7xl">
  1357→            <Skeleton className="h-48 sm:h-64 rounded-2xl mb-8" />
  1358→            <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
  1359→              {[1, 2, 3, 4].map(i => <Skeleton key={i} className="h-24 rounded-xl" />)}
  1360→            </div>
  1361→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
  1362→              {[1, 2, 3].map(i => <CardSkeleton key={i} />)}
  1363→            </div>
  1364→          </div>
  1365→        </div>
  1366→      </div>
  1367→    )
  1368→  }
  1369→
  1370→  if (error || !data) {
  1371→    return <ErrorState onRetry={fetchData} />
  1372→  }
  1373→
  1374→  const renderPage = () => {
  1375→    switch (activePage) {
  1376→      case 'home':
  1377→        return (
  1378→          <>
  1379→            <HeroBanner data={data} />
  1380→            <BrowseTiles data={data} onNavigate={setActivePage} />
  1381→            <FeaturedSection data={data} onNavigate={setActivePage} onLoginClick={openAuthDialog} />
  1382→            <CoursesSection data={data} onNavigate={setActivePage} onLoginClick={openAuthDialog} />
  1383→          </>
  1384→        )
  1385→      case 'courses':
  1386→        return <CoursesListPage data={data} onLoginClick={openAuthDialog} />
  1387→      case 'test-series':
  1388→        return <TestSeriesListPage data={data} onLoginClick={openAuthDialog} />
  1389→      case 'docs':
  1390→        return <DocsPage data={data} />
  1391→      case 'quick-links':
  1392→        return <QuickLinksPage data={data} />
  1393→      case 'about':
  1394→        return <AboutPage data={data} onLoginClick={openAuthDialog} />
  1395→      default:
  1396→        return null
  1397→    }
  1398→  }
  1399→
  1400→  return (
  1401→    <div className="min-h-screen flex flex-col bg-white">
  1402→      <PublicNavbar data={data} activePage={activePage} onNavigate={setActivePage} onLoginClick={openAuthDialog} />
  1403→      <main className="flex-1">
  1404→        <AnimatePresence mode="wait">
  1405→          <motion.div key={activePage} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -10 }} transition={{ duration: 0.2 }}>
  1406→            {renderPage()}
  1407→          </motion.div>
  1408→        </AnimatePresence>
  1409→      </main>
  1410→      <Footer data={data} />
  1411→      {/* Auth Dialog — Login, Forgot Password, Reset Password */}
  1412→      <StudentAuthDialog
  1413→        open={authDialogOpen}
  1414→        onOpenChange={setAuthDialogOpen}
  1415→        initialView={authInitialView}
  1416→        orgData={data.organization}
  1417→      />
  1418→    </div>
  1419→  )

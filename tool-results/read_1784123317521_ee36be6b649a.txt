   107→   107→export function CourseContentManager({ courseId, courseTitle, onBack }: CourseContentManagerProps) {
   108→   108→  const [modules, setModules] = useState<Module[]>([])
   109→   109→  const [courseInfo, setCourseInfo] = useState<CourseInfo | null>(null)
   110→   110→  const [loading, setLoading] = useState(true)
   111→   111→  const [activeTab, setActiveTab] = useState('content')
   112→   112→  const [viewMode, setViewMode] = useState<'list' | 'grid'>('list')
   113→   113→  const [contentSearch, setContentSearch] = useState('')
   114→   114→  const [saving, setSaving] = useState(false)
   115→   115→
   116→   116→  // Module dialog
   117→   117→  const [moduleDialogOpen, setModuleDialogOpen] = useState(false)
   118→   118→  const [moduleTitle, setModuleTitle] = useState('')
   119→   119→  const [moduleDesc, setModuleDesc] = useState('')
   120→   120→  const [editingModuleId, setEditingModuleId] = useState<string | null>(null)
   121→   121→
   122→   122→  // Lesson dialog
   123→   123→  const [lessonDialogOpen, setLessonDialogOpen] = useState(false)
   124→   124→  const [lessonModuleId, setLessonModuleId] = useState('')
   125→   125→  const [editingLessonId, setEditingLessonId] = useState<string | null>(null)
   126→   126→  const [lessonData, setLessonData] = useState({
   127→   127→    title: '', type: 'video', content: '', videoUrl: '', videoDuration: '0',
   128→   128→    fileUrl: '', notes: '', isFree: false,
   129→   129→  })
   130→   130→
   131→   131→  const fetchModules = useCallback(async () => {
   132→   132→    setLoading(true)
   133→   133→    try {
   134→   134→      const res = await apiFetch(`/api/teacher/courses/${courseId}`)
   135→   135→      const data = await res.json()
   136→   136→      setModules(data.modules || [])
   137→   137→      setCourseInfo({
   138→   138→        id: courseId,
   139→   139→        title: data.title || courseTitle,
   140→   140→        description: data.description,
   141→   141→        thumbnail: data.thumbnail,
   142→   142→        price: data.price,
   143→   143→        mrp: data.mrp,
   144→   144→        status: data.status,
   145→   145→        category: data.category,
   146→   146→        level: data.level,
   147→   147→      })
   148→   148→    } catch {
   149→   149→      toast.error('Failed to load course')
   150→   150→    } finally {
   151→   151→      setLoading(false)
   152→   152→    }
   153→   153→  }, [courseId, courseTitle])
   154→   154→
   155→   155→  useEffect(() => { fetchModules() }, [fetchModules])
   156→   156→
   157→   157→  // ── Module handlers ──
   158→   158→  const handleSaveModule = async () => {
   159→   159→    if (!moduleTitle.trim()) { toast.error('Module title required'); return }
   160→   160→    setSaving(true)
   161→   161→    try {
   162→   162→      if (editingModuleId) {
   163→   163→        await apiFetch(`/api/teacher/courses/${courseId}/modules?id=${moduleId}`, {
   164→   164→          method: 'PUT', headers: { 'Content-Type': 'application/json' },
   165→   165→          body: JSON.stringify({ id: editingModuleId, title: moduleTitle, description: moduleDesc })
   166→   166→        })
   167→   167→        toast.success('Module updated')
   168→   168→      } else {
   169→   169→        await apiFetch(`/api/teacher/courses/${courseId}/modules?id=${moduleId}`, {
   170→   170→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   171→   171→          body: JSON.stringify({ title: moduleTitle, description: moduleDesc })
   172→   172→        })
   173→   173→        toast.success('Module added')
   174→   174→      }
   175→   175→      setModuleDialogOpen(false)
   176→   176→      setModuleTitle(''); setModuleDesc(''); setEditingModuleId(null)
   177→   177→      fetchModules()
   178→   178→    } catch { toast.error('Failed to save module') }
   179→   179→    finally { setSaving(false) }
   180→   180→  }
   181→   181→
   182→   182→  const deleteModule = async (moduleId: string) => {
   183→   183→    if (!confirm('Delete this module and all its lessons?')) return
   184→   184→    try {
   185→   185→      await apiFetch(`/api/teacher/courses/${courseId}/modules?id=${moduleId}`, {
   186→   186→        method: 'DELETE', headers: { 'Content-Type': 'application/json' },
   187→   187→        
   188→   188→      })
   189→   189→      toast.success('Module deleted')
   190→   190→      fetchModules()
   191→   191→    } catch { toast.error('Failed to delete module') }
   192→   192→  }
   193→   193→
   194→   194→  // ── Lesson handlers ──
   195→   195→  const openAddLesson = (moduleId: string, type: string = 'video') => {
   196→   196→    setLessonModuleId(moduleId)
   197→   197→    setLessonData({ title: '', type, content: '', videoUrl: '', videoDuration: '0', fileUrl: '', notes: '', isFree: false })
   198→   198→    setEditingLessonId(null)
   199→   199→    setLessonDialogOpen(true)
   200→   200→  }
   201→   201→
   202→   202→  const handleSaveLesson = async () => {
   203→   203→    if (!lessonData.title.trim()) { toast.error('Lesson title required'); return }
   204→   204→    setSaving(true)
   205→   205→    try {
   206→   206→      const url = `/api/teacher/courses/${courseId}/modules/lessons?moduleId=${lessonModuleId}`
   207→   207→      const payload = { ...lessonData, videoDuration: parseInt(lessonData.videoDuration) || 0 }
   208→   208→
   209→   209→      if (editingLessonId) {
   210→   210→        await apiFetch(url, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ lessonId: editingLessonId, ...payload }) })
   211→   211→        toast.success('Lesson updated')
   212→   212→      } else {
   213→   213→        await apiFetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) })
   214→   214→        toast.success('Lesson added')
   215→   215→      }
   216→   216→      setLessonDialogOpen(false)
   217→   217→      setLessonData({ title: '', type: 'video', content: '', videoUrl: '', videoDuration: '0', fileUrl: '', notes: '', isFree: false })
   218→   218→      setEditingLessonId(null)
   219→   219→      fetchModules()
   220→   220→    } catch (err: any) { toast.error(err.message || 'Failed to save lesson') }
   221→   221→    finally { setSaving(false) }
   222→   222→  }
   223→   223→
   224→   224→  const deleteLesson = async (moduleId: string, lessonId: string) => {
   225→   225→    if (!confirm('Delete this lesson?')) return
   226→   226→    try {
   227→   227→      await apiFetch(`/api/teacher/courses/${courseId}/modules/lessons?moduleId=${moduleId}&lessonId=${lessonId}`, { method: 'DELETE' })
   228→   228→      toast.success('Lesson deleted')
   229→   229→      fetchModules()
   230→   230→    } catch { toast.error('Failed to delete lesson') }
   231→   231→  }
   232→   232→
   233→   233→  const toggleFree = async (moduleId: string, lesson: Lesson) => {
   234→   234→    try {
   235→   235→      await apiFetch(`/api/teacher/courses/${courseId}/modules/lessons?moduleId=${moduleId}`, {
   236→   236→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   237→   237→        body: JSON.stringify({ lessonId: lesson.id, isFree: !lesson.isFree })
   238→   238→      })
   239→   239→      fetchModules()
   240→   240→    } catch { toast.error('Failed to update') }
   241→   241→  }
   242→   242→
   243→   243→  const togglePublish = async () => {
   244→   244→    if (!courseInfo) return
   245→   245→    try {
   246→   246→      await apiFetch(`/api/teacher/courses/${courseId}`, {
   247→   247→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   248→   248→        body: JSON.stringify({ status: courseInfo.status === 'published' ? 'draft' : 'published' })
   249→   249→      })
   250→   250→      toast.success(courseInfo.status === 'published' ? 'Unpublished' : 'Published')
   251→   251→      fetchModules()
   252→   252→    } catch { toast.error('Failed') }
   253→   253→  }
   254→   254→
   255→   255→  const handleFileUpload = async (file: File, field: 'videoUrl' | 'fileUrl') => {
   256→   256→    if (!file) return
   257→   257→    try {
   258→   258→      const fd = new FormData()
   259→   259→      fd.append('file', file)
   260→   260→      const res = await apiFetch('/api/teacher/upload-image', { method: 'POST', body: fd })
   261→   261→      const data = await res.json()
   262→   262→      if (!res.ok) throw new Error(data.error)
   263→   263→      setLessonData(prev => ({ ...prev, [field]: data.url }))
   264→   264→      toast.success('File uploaded!')
   265→   265→    } catch (err: any) { toast.error(err.message || 'Upload failed') }
   266→   266→  }
   267→   267→
   268→   268→  // Get all lessons flattened for the content list
   269→   269→  const allLessons = modules.flatMap(m => m.lessons.map(l => ({ ...l, moduleTitle: m.title, moduleId: m.id })))
   270→   270→  const filteredLessons = contentSearch
   271→   271→    ? allLessons.filter(l => l.title.toLowerCase().includes(contentSearch.toLowerCase()))
   272→   272→    : allLessons
   273→   273→
   274→   274→  if (loading) {
   275→   275→    return <div className="flex items-center justify-center h-96"><Loader2 className="w-6 h-6 animate-spin text-emerald-600" /></div>
   276→   276→  }
   277→   277→
   278→   278→  return (
   279→   279→    <div className="space-y-6">
   280→   280→      {/* ── Header ── */}
   281→   281→      <div className="flex items-center justify-between gap-4">
   282→   282→        <div className="flex items-center gap-3">
   283→   283→          <Button variant="ghost" size="icon" onClick={onBack}><ArrowLeft className="w-5 h-5" /></Button>
   284→   284→          <div>
   285→   285→            <h1 className="text-xl font-bold">{courseTitle}</h1>
   286→   286→            <p className="text-sm text-muted-foreground">{courseInfo?.category} · {courseInfo?.level}</p>
   287→   287→          </div>
   288→   288→        </div>
   289→   289→        <div className="flex gap-2">
   290→   290→          <Button variant="outline" size="sm" className="gap-1" onClick={() => toast.info('Preview opened')}>
   291→   291→            <Eye className="w-4 h-4" /> Preview
   292→   292→          </Button>
   293→   293→          <Button
   294→   294→            size="sm"
   295→   295→            variant={courseInfo?.status === 'published' ? 'default' : 'outline'}
   296→   296→            onClick={togglePublish}
   297→   297→            className={courseInfo?.status === 'published' ? 'bg-emerald-600 hover:bg-emerald-700' : ''}
   298→   298→          >
   299→   299→            {courseInfo?.status === 'published' ? <EyeOff className="w-4 h-4 mr-1" /> : <Eye className="w-4 h-4 mr-1" />}
   300→   300→            {courseInfo?.status === 'published' ? 'Unpublish' : 'Publish'}
   301→   301→          </Button>
   302→   302→        </div>
   303→   303→      </div>
   304→   304→
   305→   305→      {/* ── Tabs ── */}
   306→   306→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   307→   307→        <TabsList>
   308→   308→          <TabsTrigger value="overview">Overview</TabsTrigger>
   309→   309→          <TabsTrigger value="content">Content</TabsTrigger>
   310→   310→          <TabsTrigger value="forum" disabled>Forum</TabsTrigger>
   311→   311→          <TabsTrigger value="chat" disabled>Chat</TabsTrigger>
   312→   312→          <TabsTrigger value="posts" disabled>Posts</TabsTrigger>
   313→   313→        </TabsList>
   314→   314→
   315→   315→        {/* ── Overview Tab ── */}
   316→   316→        <TabsContent value="overview" className="space-y-4">
   317→   317→          <div className="grid grid-cols-3 gap-4">
   318→   318→            <Card><CardContent className="p-4 flex items-center gap-3">
   319→   319→              <div className="w-10 h-10 rounded-lg bg-emerald-50 dark:bg-emerald-950/30 flex items-center justify-center">
   320→   320→                <Layers className="w-5 h-5 text-emerald-600" />
   321→   321→              </div>
   322→   322→              <div><p className="text-2xl font-bold">{modules.length}</p><p className="text-xs text-muted-foreground">Modules</p></div>
   323→   323→            </CardContent></Card>
   324→   324→            <Card><CardContent className="p-4 flex items-center gap-3">
   325→   325→              <div className="w-10 h-10 rounded-lg bg-blue-50 dark:bg-blue-950/30 flex items-center justify-center">
   326→   326→                <BookOpen className="w-5 h-5 text-blue-600" />
   327→   327→              </div>
   328→   328→              <div><p className="text-2xl font-bold">{allLessons.length}</p><p className="text-xs text-muted-foreground">Lessons</p></div>
   329→   329→            </CardContent></Card>
   330→   330→            <Card><CardContent className="p-4 flex items-center gap-3">
   331→   331→              <div className="w-10 h-10 rounded-lg bg-amber-50 dark:bg-amber-950/30 flex items-center justify-center">
   332→   332→                <Clock className="w-5 h-5 text-amber-600" />
   333→   333→              </div>
   334→   334→              <div><p className="text-2xl font-bold">{Math.floor(allLessons.reduce((s, l) => s + (l.videoDuration || 0), 0) / 60)}m</p><p className="text-xs text-muted-foreground">Duration</p></div>
   335→   335→            </CardContent></Card>
   336→   336→          </div>
   337→   337→
   338→   338→          {courseInfo?.description && (
   339→   339→            <Card><CardContent className="p-4">
   340→   340→              <h3 className="font-semibold mb-2">Course Description</h3>
   341→   341→              <p className="text-sm text-muted-foreground whitespace-pre-wrap">{courseInfo.description}</p>
   342→   342→            </CardContent></Card>
   343→   343→          )}
   344→   344→
   345→   345→          <Card><CardContent className="p-4">
   346→   346→            <h3 className="font-semibold mb-3">Modules Overview</h3>
   347→   347→            {modules.length === 0 ? (
   348→   348→              <p className="text-sm text-muted-foreground">No modules yet. Go to Content tab to add modules.</p>
   349→   349→            ) : (
   350→   350→              <div className="space-y-2">
   351→   351→                {modules.map((m, i) => (
   352→   352→                  <div key={m.id} className="flex items-center justify-between p-2 rounded-lg bg-muted/30">
   353→   353→                    <div className="flex items-center gap-3">
   354→   354→                      <div className="w-8 h-8 rounded-lg bg-emerald-100 dark:bg-emerald-950/40 flex items-center justify-center text-emerald-600 font-semibold text-sm">{i + 1}</div>
   355→   355→                      <div>
   356→   356→                        <p className="font-medium text-sm">{m.title}</p>
   357→   357→                        <p className="text-xs text-muted-foreground">{m.lessons.length} lessons</p>
   358→   358→                      </div>
   359→   359→                    </div>
   360→   360→                    <Button variant="outline" size="sm" onClick={() => { setActiveTab('content') }}>Manage</Button>
   361→   361→                  </div>
   362→   362→                ))}
   363→   363→              </div>
   364→   364→            )}
   365→   365→          </CardContent></Card>
   366→   366→        </TabsContent>
   367→   367→
   368→   368→        {/* ── Content Tab ── */}
   369→   369→        <TabsContent value="content">
   370→   370→          <div className="flex gap-6">
   371→   371→            {/* Main Content Area */}
   372→   372→            <div className="flex-1 space-y-4">
   373→   373→              {/* Toolbar */}
   374→   374→              <div className="flex items-center justify-between gap-4">
   375→   375→                <div className="flex items-center gap-2 flex-1">
   376→   376→                  <div className="relative flex-1 max-w-xs">
   377→   377→                    <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   378→   378→                    <Input placeholder="Search content..." value={contentSearch} onChange={(e) => setContentSearch(e.target.value)} className="pl-9" />
   379→   379→                  </div>
   380→   380→                  <div className="flex gap-1">
   381→   381→                    <Button variant={viewMode === 'list' ? 'default' : 'outline'} size="icon" onClick={() => setViewMode('list')}>

    86→function getTodayDate(): string {
    87→  const now = new Date()
    88→  return now.toLocaleDateString('en-IN', {
    89→    weekday: 'long',
    90→    year: 'numeric',
    91→    month: 'long',
    92→    day: 'numeric',
    93→  })
    94→}
    95→
    96→function formatCurrency(value: number): string {
    97→  if (value >= 10000000) return `₹${(value / 10000000).toFixed(2)} Cr`
    98→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)} L`
    99→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}K`
   100→  return `₹${value.toLocaleString('en-IN')}`
   101→}
   102→
   103→function formatNumber(num: number): string {
   104→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`
   105→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`
   106→  return num.toLocaleString('en-IN')
   107→}
   108→
   109→function getTimeAgo(dateStr: string): string {
   110→  const now = new Date()
   111→  const date = new Date(dateStr)
   112→  const diffMs = now.getTime() - date.getTime()
   113→  const diffMins = Math.floor(diffMs / 60000)
   114→  if (diffMins < 1) return 'just now'
   115→  if (diffMins < 60) return `${diffMins}m ago`
   116→  const diffHours = Math.floor(diffMins / 60)
   117→  if (diffHours < 24) return `${diffHours}h ago`
   118→  const diffDays = Math.floor(diffHours / 24)
   119→  return `${diffDays}d ago`
   120→}

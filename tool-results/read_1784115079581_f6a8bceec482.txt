    99→function formatCurrency(amount: number): string {
   100→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`
   101→  if (amount >= 100000) return `₹${(amount / 100000).toFixed(2)} L`
   102→  if (amount >= 1000) return `₹${(amount / 1000).toFixed(1)}K`
   103→  return `₹${amount.toLocaleString('en-IN')}`
   104→}
   105→
   106→function formatNumber(num: number): string {
   107→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`
   108→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`
   109→  return num.toLocaleString('en-IN')
   110→}

   112→function getInitials(name: string): string {
   113→  return name
   114→    .split(' ')
   115→    .map((n) => n[0])
   116→    .join('')
   117→    .slice(0, 2)
   118→    .toUpperCase()
   119→}

   143→function getTimeAgo(dateStr: string): string {
   144→  const now = new Date()
   145→  const date = new Date(dateStr)
   146→  const diffMs = now.getTime() - date.getTime()
   147→  const diffMins = Math.floor(diffMs / 60000)
   148→  if (diffMins < 1) return 'just now'
   149→  if (diffMins < 60) return `${diffMins}m ago`
   150→  const diffHours = Math.floor(diffMins / 60)
   151→  if (diffHours < 24) return `${diffHours}h ago`
   152→  const diffDays = Math.floor(diffHours / 24)
   153→  if (diffDays < 30) return `${diffDays}d ago`
   154→  return date.toLocaleDateString('en-IN', { month: 'short', day: 'numeric' })
   155→}

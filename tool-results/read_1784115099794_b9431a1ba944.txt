    93→function formatDuration(seconds: number): string {
    94→  if (!seconds) return '0m'
    95→  const h = Math.floor(seconds / 3600)
    96→  const m = Math.floor((seconds % 3600) / 60)
    97→  if (h > 0) return `${h}h ${m}m`
    98→  return `${m}m`
    99→}

   123→function extractYouTubeId(url: string): string | null {
   124→  const patterns = [
   125→    /(?:youtube\.com\/watch\?v=|youtu\.be\/|youtube\.com\/embed\/)([a-zA-Z0-9_-]{11})/,
   126→    /youtube\.com\/v\/([a-zA-Z0-9_-]{11})/,
   127→  ]
   128→  for (const p of patterns) {
   129→    const m = url.match(p)
   130→    if (m) return m[1]
   131→  }
   132→  return null
   133→}

   135→// ── Video Player Component ───────────────────────────────────────────────
   136→function VideoPlayer({ url, title }: { url: string; title: string }) {
   137→  const youtubeId = extractYouTubeId(url)

   144→          title={title}
   145→          className="absolute inset-0 w-full h-full"
   146→          allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
   147→          allowFullScreen
   148→        />
   149→      </div>

   154→    <div className="relative w-full aspect-video bg-black rounded-xl overflow-hidden shadow-lg">
   155→      <video
   156→        src={url}
   157→        controls
   158→        className="absolute inset-0 w-full h-full"
   159→        title={title}

   301→export default function CourseDetail() {
   302→  const { selectedCourseId, setStudentPage } = useAppStore()
   303→  const [course, setCourse] = useState<CourseData | null>(null)
   304→  const [purchase, setPurchase] = useState<{ id: string; purchasedAt: string; expiresAt: string | null } | null>(null)
   305→  const [progress, setProgress] = useState<Record<string, ProgressEntry>>({})
   306→  const [stats, setStats] = useState<CourseStats>({ totalLessons: 0, completedLessons: 0, totalDuration: 0, progressPercent: 0 })
   307→  const [loading, setLoading] = useState(true)
   308→  const [error, setError] = useState<string | null>(null)
   309→
   310→  // Active lesson
   311→  const [activeLessonId, setActiveLessonId] = useState<string | null>(null)
   312→  const [activeModuleId, setActiveModuleId] = useState<string | null>(null)
   313→
   314→  // Sidebar state for mobile
   315→  const [sidebarOpen, setSidebarOpen] = useState(false)
   316→
   317→  // Expanded modules
   318→  const [expandedModules, setExpandedModules] = useState<Record<string, boolean>>({})
   319→
   320→  // Marking complete
   321→  const [markingComplete, setMarkingComplete] = useState(false)
   322→
   323→  // Load course data
   324→  const load = useCallback(async () => {
   325→    if (!selectedCourseId) return
   326→    setLoading(true)
   327→    setError(null)
   328→    try {
   329→      const res = await apiFetchJSON<{
   330→        success: boolean
   331→        course: CourseData
   332→        purchase: { id: string; purchasedAt: string; expiresAt: string | null }
   333→        progress: Record<string, ProgressEntry>
   334→        stats: CourseStats
   335→      }>(`/api/student/courses/${selectedCourseId}`)
   336→      if (res.success) {
   337→        setCourse(res.course)
   338→        setPurchase(res.purchase)
   339→        setProgress(res.progress || {})
   340→        setStats(res.stats || { totalLessons: 0, completedLessons: 0, totalDuration: 0, progressPercent: 0 })
   341→
   342→        // Auto-select first lesson
   343→        if (res.course.modules.length > 0 && res.course.modules[0].lessons.length > 0) {
   344→          // Find first incomplete lesson, or first lesson
   345→          let firstIncomplete: Lesson | null = null
   346→          for (const mod of res.course.modules) {
   347→            for (const lsn of mod.lessons) {
   348→              if (res.progress[lsn.id]?.status !== 'completed') {
   349→                firstIncomplete = lsn
   350→                break
   351→              }
   352→            }
   353→            if (firstIncomplete) break
   354→          }
   355→          const targetLesson = firstIncomplete || res.course.modules[0].lessons[0]
   356→          setActiveLessonId(targetLesson.id)
   357→          setActiveModuleId(res.course.modules[0].id)
   358→          // Expand all modules that have lessons
   359→          const expanded: Record<string, boolean> = {}
   360→          res.course.modules.forEach(m => { expanded[m.id] = true })
   361→          setExpandedModules(expanded)
   362→        }
   363→      }
   364→    } catch (err) {
   365→      console.error('Course detail error:', err)
   366→      setError('Failed to load course details. Please try again.')
   367→    } finally {
   368→      setLoading(false)
   369→    }
   370→  }, [selectedCourseId])
   371→
   372→  useEffect(() => { load() }, [load])
   373→
   374→  // Find active lesson object
   375→  const activeLesson = course?.modules
   376→    .flatMap(m => m.lessons)
   377→    .find(l => l.id === activeLessonId) || null
   378→
   379→  const activeModule = course?.modules.find(m => m.id === activeModuleId) || null
   380→
   381→  // Mark lesson complete
   382→  const handleMarkComplete = async () => {
   383→    if (!activeLessonId || !selectedCourseId || markingComplete) return
   384→    setMarkingComplete(true)
   385→    try {
   386→      const res = await apiFetch('/api/student/lesson-progress', {
   387→        method: 'POST',
   388→        headers: { 'Content-Type': 'application/json' },
   389→        body: JSON.stringify({
   390→          lessonId: activeLessonId,
   391→          courseId: selectedCourseId,
   392→          status: 'completed',
   393→        }),
   394→      })
   395→      if (res.ok) {
   396→        const data = await res.json()
   397→        if (data.success) {
   398→          setProgress(prev => ({
   399→            ...prev,
   400→            [activeLessonId]: { status: 'completed', completedAt: new Date().toISOString() },
   401→          }))
   402→          if (data.courseStats) {
   403→            setStats(prev => ({
   404→              ...prev,
   405→              completedLessons: data.courseStats.completedLessons,
   406→              progressPercent: data.courseStats.progressPercent,
   407→            }))
   408→          }
   409→          toast.success('Lesson completed! 🎉')
   410→        }
   411→      }
   412→    } catch {
   413→      toast.error('Failed to update progress')
   414→    } finally {
   415→      setMarkingComplete(false)
   416→    }
   417→  }
   418→
   419→  // Navigate to next lesson
   420→  const handleNextLesson = () => {
   421→    if (!course) return
   422→    const allLessons = course.modules.flatMap(m => m.lessons)
   423→    const currentIndex = allLessons.findIndex(l => l.id === activeLessonId)
   424→    if (currentIndex < allLessons.length - 1) {
   425→      const next = allLessons[currentIndex + 1]
   426→      setActiveLessonId(next.id)
   427→      // Find which module this lesson belongs to
   428→      for (const mod of course.modules) {
   429→        if (mod.lessons.some(l => l.id === next.id)) {
   430→          setActiveModuleId(mod.id)
   431→          setExpandedModules(prev => ({ ...prev, [mod.id]: true }))
   432→          break
   433→        }
   434→      }
   435→    }
   436→  }
   437→
   438→  // Navigate to previous lesson
   439→  const handlePrevLesson = () => {
   440→    if (!course) return
   441→    const allLessons = course.modules.flatMap(m => m.lessons)
   442→    const currentIndex = allLessons.findIndex(l => l.id === activeLessonId)
   443→    if (currentIndex > 0) {
   444→      const prev = allLessons[currentIndex - 1]
   445→      setActiveLessonId(prev.id)
   446→      for (const mod of course.modules) {
   447→        if (mod.lessons.some(l => l.id === prev.id)) {
   448→          setActiveModuleId(mod.id)
   449→          setExpandedModules(prev2 => ({ ...prev2, [mod.id]: true }))
   450→          break
   451→        }
   452→      }
   453→    }

   464→      <div className="space-y-6">
   465→        <div className="flex items-center gap-3">
   466→          <Skeleton className="size-9 rounded-md" />
   467→          <div>
   468→            <Skeleton className="h-6 w-48 mb-1" />
   469→            <Skeleton className="h-4 w-32" />
   470→          </div>
   471→        </div>
   472→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   473→          <div className="lg:col-span-2">
   474→            <Skeleton className="aspect-video w-full rounded-xl" />
   475→            <Skeleton className="h-4 w-3/4 mt-4" />
   476→            <Skeleton className="h-4 w-1/2 mt-2" />
   477→          </div>
   478→          <Skeleton className="h-96 rounded-xl" />
   479→        </div>
   480→      </div>
   481→    )
   482→  }
   483→
   484→  // ── Error State ──
   485→  if (error) {

   488→        <div className="flex items-center gap-3">

   518→  const isPurchased = !!purchase
   519→  const allLessons = course.modules.flatMap(m => m.lessons)
   520→  const currentIndex = allLessons.findIndex(l => l.id === activeLessonId)
   521→  const hasNext = currentIndex < allLessons.length - 1
   522→  const hasPrev = currentIndex > 0
   523→
   524→  // ── Sidebar: Module & Lesson List ──
   525→  const SidebarContent = () => (
   526→    <div className="space-y-1">
   527→      {/* Course Progress */}
   528→      <div className="p-4 border-b bg-gradient-to-r from-amber-50 to-white">
   529→        <div className="flex items-center justify-between mb-2">
   530→          <span className="text-xs font-semibold text-amber-800 uppercase tracking-wide">Progress</span>

    34→// ── Lazy loader with retry for stale chunks ───────────────────────────────
    35→function lazyWithRetry<T>(importFn: () => Promise<T>, retries = 2): Promise<T> {
    36→  return importFn().catch(async (error) => {
    37→    const isChunkError =
    38→      error?.name === 'ChunkLoadError' ||
    39→      String(error?.message || '').includes('Failed to load chunk') ||
    40→      String(error?.message || '').includes('Loading chunk')
    41→    if (isChunkError && retries > 0) {
    42→      await new Promise((r) => setTimeout(r, 400))
    43→      return lazyWithRetry(importFn, retries - 1)
    44→    }
    45→    throw error
    46→  })
    47→}

    52→if (typeof window !== 'undefined') {
    53→  const w = window as unknown as { __loginChunkHandlerInstalled?: boolean }
    54→  if (!w.__loginChunkHandlerInstalled) {
    55→    w.__loginChunkHandlerInstalled = true
    56→    window.addEventListener('error', (e) => {
    57→      const msg = e?.message || ''
    58→      if (
    59→        msg.includes('Failed to load chunk') ||
    60→        msg.includes('Loading chunk') ||
    61→        msg.includes('ChunkLoadError')
    62→      ) {
    63→        e.preventDefault()
    64→        window.location.reload()
    65→      }
    66→    })
    67→    window.addEventListener('unhandledrejection', (e) => {
    68→      const reason = e?.reason
    69→      const msg = (reason?.message || String(reason || '')) as string
    70→      if (
    71→        msg.includes('Failed to load chunk') ||

    74→      ) {
    75→        e.preventDefault()
    76→        window.location.reload()
    77→      }
    78→    })
    79→  }

    93→// ─── Account Lock Countdown Component ───────────────────────────────────────
    94→function AccountLockCountdown({ minutes, onComplete }: { minutes: number; onComplete: () => void }) {
    95→  const [remaining, setRemaining] = useState(minutes * 60)
    96→
    97→  useEffect(() => {
    98→    if (remaining <= 0) {
    99→      onComplete()
   100→      return
   101→    }
   102→
   103→    const timer = setInterval(() => {
   104→      setRemaining((prev) => {
   105→        if (prev <= 1) {
   106→          onComplete()
   107→          return 0
   108→        }
   109→        return prev - 1
   110→      })
   111→    }, 1000)
   112→
   113→    return () => clearInterval(timer)
   114→  }, [remaining, onComplete])
   115→
   116→  const mins = Math.floor(remaining / 60)
   117→  const secs = remaining % 60
   118→
   119→  return (
   120→    <div className="flex items-center gap-2 p-3 rounded-xl bg-red-50 border border-red-200">
   121→      <Clock className="w-5 h-5 text-red-500 flex-shrink-0" />
   122→      <div>
   123→        <p className="text-sm font-medium text-red-700">
   124→          Account temporarily locked
   125→        </p>
   126→        <p className="text-xs text-red-500">
   127→          Try again in {mins}m {secs.toString().padStart(2, '0')}s
   128→        </p>
   129→      </div>

   222→// ─── Password Strength Indicator ────────────────────────────────────────────
   223→function PasswordStrengthIndicator({ password }: { password: string }) {
   224→  const strength = useMemo(() => {
   225→    if (!password) return { score: 0, label: '', feedback: [] as string[] }
   226→
   227→    const feedback: string[] = []
   228→    let metCount = 0
   229→
   230→    if (password.length >= 8) metCount++; else feedback.push('8+ characters')
   231→    if (/[A-Z]/.test(password)) metCount++; else feedback.push('uppercase letter')
   232→    if (/[a-z]/.test(password)) metCount++; else feedback.push('lowercase letter')
   233→    if (/\d/.test(password)) metCount++; else feedback.push('number')
   234→    if (/[^A-Za-z0-9]/.test(password)) metCount++; else feedback.push('special character')
   235→
   236→    let score: number
   237→    let label: string
   238→    if (metCount <= 0) { score = 0; label = 'very-weak' }
   239→    else if (metCount <= 2) { score = 1; label = 'weak' }
   240→    else if (metCount === 3) { score = 2; label = 'fair' }
   241→    else if (metCount === 4) { score = 3; label = 'good' }
   242→    else { score = 4; label = 'strong' }
   243→
   244→    return { score, label, feedback }
   245→  }, [password])
   246→
   247→  if (!password) return null
   248→
   249→  const colors = ['bg-red-500', 'bg-orange-500', 'bg-yellow-500', 'bg-emerald-400', 'bg-emerald-600']
   250→  const labels = ['Very Weak', 'Weak', 'Fair', 'Good', 'Strong']
   251→  const textColors = ['text-red-600', 'text-orange-600', 'text-yellow-600', 'text-emerald-500', 'text-emerald-700']
   252→
   253→  return (
   254→    <div className="space-y-2">
   255→      <div className="flex gap-1">
   256→        {[0, 1, 2, 3, 4].map((i) => (
   257→          <div
   258→            key={i}
   259→            className={`h-1.5 flex-1 rounded-full transition-all duration-300 ${
   260→              i <= strength.score ? colors[strength.score] : 'bg-gray-200'
   261→            }`}
   262→          />
   263→        ))}
   264→      </div>
   265→      <div className="flex items-center justify-between">
   266→        <span className={`text-xs font-medium ${textColors[strength.score]}`}>
   267→          {labels[strength.score]}
   268→        </span>
   269→        {strength.feedback.length > 0 && strength.score < 4 && (
   270→          <span className="text-xs text-gray-400">
   271→            Need: {strength.feedback.join(', ')}
   272→          </span>
   273→        )}
   274→      </div>
   275→    </div>
   276→  )
   277→}

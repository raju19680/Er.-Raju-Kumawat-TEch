    34→    34→// ── Lazy loader with retry for stale chunks ───────────────────────────────
    35→    35→function lazyWithRetry<T>(importFn: () => Promise<T>, retries = 2): Promise<T> {
    36→    36→  return importFn().catch(async (error) => {
    37→    37→    const isChunkError =
    38→    38→      error?.name === 'ChunkLoadError' ||
    39→    39→      String(error?.message || '').includes('Failed to load chunk') ||
    40→    40→      String(error?.message || '').includes('Loading chunk')
    41→    41→    if (isChunkError && retries > 0) {
    42→    42→      await new Promise((r) => setTimeout(r, 400))
    43→    43→      return lazyWithRetry(importFn, retries - 1)
    44→    44→    }
    45→    45→    throw error
    46→    46→  })
    47→    47→}

    54→    54→  if (!w.__loginChunkHandlerInstalled) {
    55→    55→    w.__loginChunkHandlerInstalled = true

    94→    94→function AccountLockCountdown({ minutes, onComplete }: { minutes: number; onComplete: () => void }) {
    95→    95→  const [remaining, setRemaining] = useState(minutes * 60)
    96→    96→
    97→    97→  useEffect(() => {
    98→    98→    if (remaining <= 0) {
    99→    99→      onComplete()
   100→   100→      return
   101→   101→    }
   102→   102→
   103→   103→    const timer = setInterval(() => {
   104→   104→      setRemaining((prev) => {
   105→   105→        if (prev <= 1) {
   106→   106→          onComplete()
   107→   107→          return 0
   108→   108→        }
   109→   109→        return prev - 1
   110→   110→      })
   111→   111→    }, 1000)
   112→   112→
   113→   113→    return () => clearInterval(timer)
   114→   114→  }, [remaining, onComplete])
   115→   115→
   116→   116→  const mins = Math.floor(remaining / 60)
   117→   117→  const secs = remaining % 60
   118→   118→
   119→   119→  return (
   120→   120→    <div className="flex items-center gap-2 p-3 rounded-xl bg-red-50 border border-red-200">

   222→   222→// ─── Password Strength Indicator ────────────────────────────────────────────
   223→   223→function PasswordStrengthIndicator({ password }: { password: string }) {
   224→   224→  const strength = useMemo(() => {
   225→   225→    if (!password) return { score: 0, label: '', feedback: [] as string[] }
   226→   226→
   227→   227→    const feedback: string[] = []
   228→   228→    let metCount = 0
   229→   229→
   230→   230→    if (password.length >= 8) metCount++; else feedback.push('8+ characters')
   231→   231→    if (/[A-Z]/.test(password)) metCount++; else feedback.push('uppercase letter')
   232→   232→    if (/[a-z]/.test(password)) metCount++; else feedback.push('lowercase letter')
   233→   233→    if (/\d/.test(password)) metCount++; else feedback.push('number')
   234→   234→    if (/[^A-Za-z0-9]/.test(password)) metCount++; else feedback.push('special character')
   235→   235→
   236→   236→    let score: number
   237→   237→    let label: string
   238→   238→    if (metCount <= 0) { score = 0; label = 'very-weak' }
   239→   239→    else if (metCount <= 2) { score = 1; label = 'weak' }
   240→   240→    else if (metCount === 3) { score = 2; label = 'fair' }
   241→   241→    else if (metCount === 4) { score = 3; label = 'good' }
   242→   242→    else { score = 4; label = 'strong' }
   243→   243→
   244→   244→    return { score, label, feedback }
   245→   245→  }, [password])
   246→   246→
   247→   247→  if (!password) return null
   248→   248→
   249→   249→  const colors = ['bg-red-500', 'bg-orange-500', 'bg-yellow-500', 'bg-emerald-400', 'bg-emerald-600']
   250→   250→  const labels = ['Very Weak', 'Weak', 'Fair', 'Good', 'Strong']
   251→   251→  const textColors = ['text-red-600', 'text-orange-600', 'text-yellow-600', 'text-emerald-500', 'text-emerald-700']
   252→   252→
   253→   253→  return (
   254→   254→    <div className="space-y-2">
   255→   255→      <div className="flex gap-1">
   256→   256→        {[0, 1, 2, 3, 4].map((i) => (
   257→   257→          <div
   258→   258→            key={i}
   259→   259→            className={`h-1.5 flex-1 rounded-full transition-all duration-300 ${
   260→   260→              i <= strength.score ? colors[strength.score] : 'bg-gray-200'
   261→   261→            }`}
   262→   262→          />
   263→   263→        ))}
   264→   264→      </div>
   265→   265→      <div className="flex items-center justify-between">
   266→   266→        <span className={`text-xs font-medium ${textColors[strength.score]}`}>
   267→   267→          {labels[strength.score]}
   268→   268→        </span>
   269→   269→        {strength.feedback.length > 0 && strength.score < 4 && (
   270→   270→          <span className="text-xs text-gray-400">
   271→   271→            Need: {strength.feedback.join(', ')}
   272→   272→          </span>
   273→   273→        )}
   274→   274→      </div>
   275→   275→    </div>
   276→   276→  )
   277→   277→}

   168→// ── Component ────────────────────────────────────────────────────────────────
   169→
   170→export default function TestSeriesList() {
   171→  const { orgCode } = useAppStore()
   172→
   173→  // List state
   174→  const [searchQuery, setSearchQuery] = useState('')
   175→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   176→  const [statusFilter, setStatusFilter] = useState<string>('all')
   177→  const [viewMode, setViewMode] = useState<'list' | 'grid'>('list')
   178→  const [showFilterDropdown, setShowFilterDropdown] = useState(false)
   179→  const [currentPageNum, setCurrentPageNum] = useState(1)
   180→
   181→  // API-driven state
   182→  const [series, setSeries] = useState<TestSeriesItem[]>([])
   183→  const [totalItems, setTotalItems] = useState(0)
   184→  const [loading, setLoading] = useState(true)
   185→  const [error, setError] = useState<string | null>(null)
   186→
   187→  // Drawer state
   188→  const [drawerOpen, setDrawerOpen] = useState(false)
   189→  const [activeTab, setActiveTab] = useState<string>('basic')
   190→  const [saving, setSaving] = useState(false)
   191→  const [editingId, setEditingId] = useState<string | null>(null)
   192→
   193→  // Delete dialog
   194→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   195→  const [seriesToDelete, setSeriesToDelete] = useState<TestSeriesItem | null>(null)
   196→  const [deleting, setDeleting] = useState(false)
   197→
   198→  // Form state
   199→  const [formData, setFormData] = useState<FormData>({ ...defaultFormData })
   200→
   201→  // Reference data from API
   202→  const [categories, setCategories] = useState<CategoryOption[]>([])
   203→  const [testOptions, setTestOptions] = useState<TestOption[]>([])
   204→  const [loadingTests, setLoadingTests] = useState(false)
   205→
   206→  const itemsPerPage = 20
   207→
   208→  const updateField = <K extends keyof FormData>(key: K, value: FormData[K]) => {
   209→    setFormData((prev) => ({ ...prev, [key]: value }))
   210→  }
   211→
   212→  const resetForm = () => {
   213→    setFormData({ ...defaultFormData })
   214→    setActiveTab('basic')
   215→    setEditingId(null)
   216→  }
   217→
   218→  // ── Fetch test series list ───────────────────────────────────────────────
   219→
   220→  const fetchSeries = useCallback(async () => {
   221→    if (!orgCode) return
   222→    setLoading(true)
   223→    setError(null)
   224→    try {
   225→      const params = new URLSearchParams({
   226→        search: searchQuery,
   227→        page: String(currentPageNum),
   228→        limit: String(itemsPerPage),
   229→        organizationId: orgCode,
   230→      })
   231→      if (categoryFilter !== 'all') params.set('category', categoryFilter)
   232→      if (statusFilter !== 'all') params.set('status', statusFilter)
   233→
   234→      const data = await apiFetchJSON<PaginatedResponse<TestSeriesItem>>(
   235→        `/api/teacher/test-series?${params.toString()}`
   236→      )
   237→      setSeries(data.items || [])
   238→      setTotalItems(data.total || 0)
   239→    } catch (err) {
   240→      console.error('Failed to fetch test series:', err)
   241→      setError('Failed to load test series. Please try again.')
   242→    } finally {
   243→      setLoading(false)
   244→    }
   245→  }, [orgCode, searchQuery, currentPageNum, categoryFilter, statusFilter])
   246→
   247→  // ── Fetch categories ─────────────────────────────────────────────────────
   248→
   249→  const fetchCategories = useCallback(async () => {
   250→    if (!orgCode) return
   251→    try {
   252→      const data = await apiFetchJSON<{ items: CategoryOption[] }>(
   253→        `/api/categories?organizationId=${orgCode}&limit=100`
   254→      )
   255→      setCategories(data.items || [])
   256→    } catch {
   257→      // Silently fail
   258→    }
   259→  }, [orgCode])
   260→
   261→  // ── Fetch test options for combo selection ───────────────────────────────
   262→
   263→  const fetchTestOptions = useCallback(async () => {
   264→    if (!orgCode) return
   265→    setLoadingTests(true)
   266→    try {
   267→      const data = await apiFetchJSON<PaginatedResponse<TestOption>>(
   268→        `/api/tests?organizationId=${orgCode}&limit=100`
   269→      )
   270→      setTestOptions(data.items || [])
   271→    } catch {
   272→      // Silently fail
   273→    } finally {
   274→      setLoadingTests(false)
   275→    }
   276→  }, [orgCode])
   277→
   278→  useEffect(() => {
   279→    fetchSeries()
   280→  }, [fetchSeries])
   281→
   282→  useEffect(() => {
   283→    fetchCategories()
   284→  }, [fetchCategories])
   285→
   286→  // Load test options when drawer opens with combo enabled
   287→  useEffect(() => {
   288→    if (drawerOpen) {
   289→      fetchTestOptions()
   290→    }
   291→  }, [drawerOpen, fetchTestOptions])
   292→
   293→  const totalPages = Math.max(1, Math.ceil(totalItems / itemsPerPage))
   294→
   295→  const discount = (price: number, mrp: number) =>
   296→    mrp > price ? Math.round(((mrp - price) / mrp) * 100) : 0
   297→
   298→  // ── Delete handler ───────────────────────────────────────────────────────
   299→
   300→  const handleDelete = async () => {
   301→    if (!seriesToDelete) return
   302→    setDeleting(true)
   303→    try {
   304→      await apiFetchJSON(`/api/teacher/test-series/${seriesToDelete.id}`, { method: 'DELETE' })
   305→      toast.success('Test series deleted successfully')
   306→      setDeleteDialogOpen(false)
   307→      setSeriesToDelete(null)
   308→      fetchSeries()
   309→    } catch {
   310→      toast.error('Failed to delete test series')
   311→    } finally {
   312→      setDeleting(false)
   313→    }
   314→  }
   315→
   316→  // ── Open drawer for edit ─────────────────────────────────────────────────
   317→
   318→  const openEditDrawer = (item: TestSeriesItem) => {
   319→    setEditingId(item.id)
   320→    setFormData({
   321→      title: item.title || '',
   322→      description: (item.description as string) || '',
   323→      category: item.category || '',
   324→      price: item.price?.toString() || '',
   325→      mrp: item.mrp?.toString() || '',
   326→      validityMode: (item.validityMode as FormData['validityMode']) || 'days',
   327→      validityDays: (item.validityDays as string)?.toString() || '',
   328→      validityDate: (item.validityDate as string) || '',
   329→      isCombo: item.isCombo || false,
   330→      selectedTests: (item.selectedTests as string[]) || [],
   331→      includeTestMaker: (item.includeTestMaker as boolean) || false,
   332→      allowPayment: (item.allowPayment as boolean) ?? true,
   333→      seoTitle: (item.seoTitle as string) || '',
   334→      seoDescription: (item.seoDescription as string) || '',
   335→      richSnippets: (item.richSnippets as boolean) || false,
   336→      imagePreview: item.thumbnail || null,
   337→      status: item.status || 'draft',
   338→      sortOrder: item.sortOrder?.toString() || '',
   339→    })
   340→    setActiveTab('basic')
   341→    setDrawerOpen(true)
   342→  }
   343→
   344→  // ── Submit handler ───────────────────────────────────────────────────────
   345→
   346→  const handleSubmit = async () => {
   347→    if (!formData.title.trim()) {
   348→      toast.error('Title is required')
   349→      return
   350→    }
   351→
   352→    setSaving(true)
   353→    try {
   354→      const body: Record<string, unknown> = {
   355→        title: formData.title,
   356→        description: formData.description,
   357→        category: formData.category,
   358→        price: formData.price ? Number(formData.price) : 0,
   359→        mrp: formData.mrp ? Number(formData.mrp) : 0,
   360→        validityMode: formData.validityMode,
   361→        validityDays: formData.validityDays ? Number(formData.validityDays) : undefined,
   362→        validityDate: formData.validityDate || undefined,
   363→        isCombo: formData.isCombo,
   364→        selectedTests: formData.isCombo ? formData.selectedTests : [],
   365→        includeTestMaker: formData.includeTestMaker,
   366→        allowPayment: formData.allowPayment,
   367→        seoTitle: formData.seoTitle || undefined,
   368→        seoDescription: formData.seoDescription || undefined,
   369→        richSnippets: formData.richSnippets,
   370→        status: formData.status,
   371→        sortOrder: formData.sortOrder ? Number(formData.sortOrder) : undefined,
   372→        thumbnail: formData.imagePreview || undefined,
   373→        organizationId: orgCode,
   374→      }
   375→
   376→      if (editingId) {
   377→        await apiFetchJSON(`/api/teacher/test-series/${editingId}`, {
   378→          method: 'PUT',
   379→          body: JSON.stringify(body),
   380→        })
   381→        toast.success('Test series updated successfully')
   382→      } else {
   383→        await apiFetchJSON('/api/teacher/test-series', {
   384→          method: 'POST',
   385→          body: JSON.stringify(body),
   386→        })
   387→        toast.success('Test series created successfully')
   388→      }
   389→
   390→      setDrawerOpen(false)
   391→      resetForm()
   392→      fetchSeries()
   393→    } catch (err) {
   394→      console.error('Failed to save test series:', err)
   395→      toast.error(editingId ? 'Failed to update test series' : 'Failed to create test series')
   396→    } finally {
   397→      setSaving(false)
   398→    }
   399→  }
   400→
   401→  // ── Render ───────────────────────────────────────────────────────────────
   402→
   403→  return (
   404→    <div className="space-y-5">
   405→      {/* ── Header row ──────────────────────────────────────────────────── */}
   406→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   407→        <h1 className="text-xl sm:text-2xl font-semibold tracking-tight">Test Series</h1>
   408→        <Button
   409→          className="w-full sm:w-fit"
   410→          onClick={() => {
   411→            resetForm()
   412→            setDrawerOpen(true)
   413→          }}
   414→        >
   415→          <Plus className="mr-2 h-4 w-4" />
   416→          Add Test Series
   417→        </Button>
   418→      </div>
   419→
   420→      {/* ── Search / Filter / View toggle bar ───────────────────────────── */}
   421→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   422→        {/* Search */}
   423→        <div className="relative flex-1">
   424→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   425→          <Input
   426→            placeholder="Search test series..."
   427→            value={searchQuery}
   428→            onChange={(e) => {
   429→              setSearchQuery(e.target.value)
   430→              setCurrentPageNum(1)
   431→            }}
   432→            className="pl-9 rounded-lg"
   433→          />
   434→        </div>
   435→
   436→        {/* Filter */}
   437→        <div className="relative">
   438→          <Button
   439→            variant="outline"
   440→            size="sm"
   441→            className="gap-2"
   442→            onClick={() => setShowFilterDropdown(!showFilterDropdown)}
   443→          >
   444→            <Filter className="h-4 w-4" />
   445→            Filters
   446→            {(categoryFilter !== 'all' || statusFilter !== 'all') && (
   447→              <span className="flex h-5 w-5 items-center justify-center rounded-full bg-primary text-[10px] text-primary-foreground">
   448→                {(categoryFilter !== 'all' ? 1 : 0) + (statusFilter !== 'all' ? 1 : 0)}
   449→              </span>
   450→            )}
   451→          </Button>
   452→
   453→          {showFilterDropdown && (
   454→            <div className="absolute right-0 top-full z-50 mt-2 w-72 rounded-xl border bg-white p-4 shadow-lg">
   455→              <div className="mb-3 flex items-center justify-between">
   456→                <span className="text-sm font-semibold">Filters</span>
   457→                <button
   458→                  onClick={() => setShowFilterDropdown(false)}
   459→                  className="text-muted-foreground hover:text-foreground"
   460→                >
   461→                  <X className="h-4 w-4" />
   462→                </button>
   463→              </div>
   464→              <div className="space-y-3">
   465→                <div>
   466→                  <label className="mb-1 block text-xs font-medium text-muted-foreground">
   467→                    Category
   468→                  </label>
   469→                  <Select value={categoryFilter} onValueChange={(v) => { setCategoryFilter(v); setCurrentPageNum(1) }}>
   470→                    <SelectTrigger className="w-full">
   471→                      <SelectValue />
   472→                    </SelectTrigger>
   473→                    <SelectContent>
   474→                      <SelectItem value="all">All Categories</SelectItem>
   475→                      {categories.length > 0
   476→                        ? categories.map((c) => (
   477→                            <SelectItem key={c.id} value={c.name}>
   478→                              {c.name}
   479→                            </SelectItem>
   480→                          ))
   481→                        : ['JEE', 'NEET', 'CUET', 'GATE', 'UPSC', 'CAT', 'Other'].map((c) => (
   482→                            <SelectItem key={c} value={c}>
   483→                              {c}
   484→                            </SelectItem>
   485→                          ))
   486→                      }
   487→                    </SelectContent>
   488→                  </Select>
   489→                </div>
   490→                <div>
   491→                  <label className="mb-1 block text-xs font-medium text-muted-foreground">
   492→                    Status
   493→                  </label>
   494→                  <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setCurrentPageNum(1) }}>
   495→                    <SelectTrigger className="w-full">
   496→                      <SelectValue />
   497→                    </SelectTrigger>
   498→                    <SelectContent>
   499→                      <SelectItem value="all">All Status</SelectItem>
   500→                      <SelectItem value="published">Published</SelectItem>
   501→                      <SelectItem value="draft">Draft</SelectItem>
   502→                    </SelectContent>
   503→                  </Select>
   504→                </div>
   505→                <Button
   506→                  variant="ghost"
   507→                  size="sm"
   508→                  className="w-full"
   509→                  onClick={() => {
   510→                    setCategoryFilter('all')
   511→                    setStatusFilter('all')
   512→                    setCurrentPageNum(1)
   513→                  }}
   514→                >
   515→                  Clear Filters
   516→                </Button>
   517→              </div>
   518→            </div>
   519→          )}
   520→        </div>
   521→
   522→        {/* View toggle */}
   523→        <div className="flex items-center rounded-lg border bg-muted p-0.5">
   524→          <button
   525→            onClick={() => setViewMode('list')}
   526→            className={`flex items-center justify-center rounded-md p-1.5 transition-colors ${
   527→              viewMode === 'list'
   528→                ? 'bg-background text-foreground shadow-sm'
   529→                : 'text-muted-foreground hover:text-foreground'
   530→            }`}
   531→          >
   532→            <List className="h-4 w-4" />
   533→          </button>
   534→          <button
   535→            onClick={() => setViewMode('grid')}
   536→            className={`flex items-center justify-center rounded-md p-1.5 transition-colors ${
   537→              viewMode === 'grid'
   538→                ? 'bg-background text-foreground shadow-sm'
   539→                : 'text-muted-foreground hover:text-foreground'
   540→            }`}
   541→          >
   542→            <LayoutGrid className="h-4 w-4" />
   543→          </button>
   544→        </div>
   545→      </div>
   546→
   547→      {/* ── Error state ────────────────────────────────────────────────── */}
   548→      {error && (
   549→        <div className="flex items-center gap-3 rounded-lg border border-destructive/30 bg-destructive/5 px-4 py-3 text-sm text-destructive">
   550→          <AlertCircle className="h-4 w-4 flex-shrink-0" />
   551→          <span>{error}</span>
   552→          <Button variant="ghost" size="sm" className="ml-auto" onClick={fetchSeries}>
   553→            Retry
   554→          </Button>
   555→        </div>
   556→      )}
   557→
   558→      {/* ── Loading skeletons ────────────────────────────────────────────── */}
   559→      {loading ? (
   560→        <div className="space-y-3">
   561→          {Array.from({ length: 4 }).map((_, i) => (
   562→            <Card key={i} className="overflow-hidden rounded-xl">
   563→              <CardContent className="p-4">
   564→                <div className="flex items-center gap-4">
   565→                  <Skeleton className="h-14 w-14 rounded-lg" />
   566→                  <div className="flex-1 space-y-2">
   567→                    <Skeleton className="h-4 w-48" />
   568→                    <Skeleton className="h-3 w-32" />
   569→                  </div>
   570→                  <Skeleton className="h-4 w-16" />
   571→                  <Skeleton className="h-6 w-20" />
   572→                </div>
   573→              </CardContent>
   574→            </Card>
   575→          ))}
   576→        </div>
   577→      ) : series.length === 0 ? (
   578→      /* ── Empty state ────────────────────────────────────────────────────── */
   579→        <div className="flex items-center justify-center rounded-xl border bg-white py-16">
   580→          <div className="text-center">
   581→            <Package className="mx-auto h-12 w-12 text-muted-foreground/40" />
   582→            <p className="mt-3 text-sm text-muted-foreground">No test series found.</p>
   583→          </div>
   584→        </div>
   585→      ) : viewMode === 'list' ? (
   586→        /* List view */
   587→        <div className="space-y-3">
   588→          {series.map((item) => (
   589→            <Card key={item.id} className="overflow-hidden rounded-xl">
   590→              <CardContent className="p-0">
   591→                <div className="flex flex-col sm:flex-row items-start gap-3 sm:gap-4 p-4">
   592→                  {/* Thumbnail */}
   593→                  <div
   594→                    className={`flex h-14 w-14 flex-shrink-0 items-center justify-center rounded-lg text-white ${
   595→                      categoryColors[item.category] || 'bg-slate-500'
   596→                    }`}
   597→                  >
   598→                    <span className="text-lg font-bold">{(item.category || '?').charAt(0)}</span>
   599→                  </div>
   600→
   601→                  {/* Info */}
   602→                  <div className="min-w-0 flex-1">

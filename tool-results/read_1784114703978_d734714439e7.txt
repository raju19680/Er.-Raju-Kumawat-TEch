    85→
    86→  // Bulk actions dropdown

    88→
    89→  // Wizard state
    90→  const [wizardOpen, setWizardOpen] = useState(false)
    91→  const [editingCourse, setEditingCourse] = useState<Course | null>(null)
    92→  const [currentStep, setCurrentStep] = useState(1)
    93→  const [saving, setSaving] = useState(false)
    94→
    95→  // Delete state
    96→  const [courseToDelete, setCourseToDelete] = useState<Course | null>(null)
    97→
    98→  // Content management state
    99→  const [contentManagerCourse, setContentManagerCourse] = useState<Course | null>(null)
   100→
   101→  // Row dropdown state

   106→

   138→        category: course.category || '', language: course.language || 'Hindi', level: course.level || 'All',
   139→        featured: course.featured || false, status: course.status || 'draft',
   140→        validityType: 'lifetime', validityMonths: '12', validityEndDate: '', discountCode: '', sortOrder: '0',

   157→      const payload = {
   158→        ...formData,
   159→        price: parseFloat(formData.price) || 0,

   161→        validityMonths: formData.validityType === 'months' ? parseInt(formData.validityMonths) : null,

   254→                        <item.icon className={`w-3.5 h-3.5 ${item.color}`} />
   255→                        <span>{item.label}</span>
   256→                      </button>
   257→                    ))}

   259→                </div>

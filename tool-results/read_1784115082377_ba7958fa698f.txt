    99→    99→function formatCurrency(amount: number): string {
   100→   100→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)} Cr`
   101→   101→  if (amount >= 100000) return `₹${(amount / 100000).toFixed(2)} L`
   102→   102→  if (amount >= 1000) return `₹${(amount / 1000).toFixed(1)}K`
   103→   103→  return `₹${amount.toLocaleString('en-IN')}`
   104→   104→}
   105→   105→
   106→   106→function formatNumber(num: number): string {
   107→   107→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`
   108→   108→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`
   109→   109→  return num.toLocaleString('en-IN')
   110→   110→}

   143→   143→function getTimeAgo(dateStr: string): string {
   144→   144→  const now = new Date()
   145→   145→  const date = new Date(dateStr)
   146→   146→  const diffMs = now.getTime() - date.getTime()
   147→   147→  const diffMins = Math.floor(diffMs / 60000)
   148→   148→  if (diffMins < 1) return 'just now'
   149→   149→  if (diffMins < 60) return `${diffMins}m ago`
   150→   150→  const diffHours = Math.floor(diffMins / 60)
   151→   151→  if (diffHours < 24) return `${diffHours}h ago`
   152→   152→  const diffDays = Math.floor(diffHours / 24)
   153→   153→  if (diffDays < 30) return `${diffDays}d ago`
   154→   154→  return date.toLocaleDateString('en-IN', { month: 'short', day: 'numeric' })
   155→   155→}

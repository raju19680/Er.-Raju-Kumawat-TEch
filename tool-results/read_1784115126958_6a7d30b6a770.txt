   134→// ─── Component ────────────────────────────────────────────────────────────
   135→
   136→export default function DigitalProductsList() {
   137→  const { orgCode } = useAppStore()
   138→
   139→  // Data state
   140→  const [products, setProducts] = useState<DigitalProduct[]>([])
   141→  const [categories, setCategories] = useState<Category[]>([])
   142→  const [total, setTotal] = useState(0)
   143→  const [totalPages, setTotalPages] = useState(1)
   144→
   145→  // UI state
   146→  const [loading, setLoading] = useState(true)
   147→  const [error, setError] = useState<string | null>(null)
   148→  const [search, setSearch] = useState('')
   149→  const [searchDebounced, setSearchDebounced] = useState('')
   150→  const [statusFilter, setStatusFilter] = useState<string>('all')
   151→  const [typeFilter, setTypeFilter] = useState<string>('all')
   152→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   153→  const [page, setPage] = useState(1)
   154→  const limit = 20
   155→
   156→  // Drawer state
   157→  const [drawerOpen, setDrawerOpen] = useState(false)
   158→  const [editProduct, setEditProduct] = useState<DigitalProduct | null>(null)
   159→  const [saving, setSaving] = useState(false)
   160→
   161→  // Delete dialog state
   162→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   163→  const [productToDelete, setProductToDelete] = useState<DigitalProduct | null>(null)
   164→  const [deleting, setDeleting] = useState(false)
   165→
   166→  // Duplicate dialog state
   167→  const [duplicating, setDuplicating] = useState(false)
   168→
   169→  // Form state
   170→  const [formTitle, setFormTitle] = useState('')
   171→  const [formDescription, setFormDescription] = useState('')
   172→  const [formThumbnail, setFormThumbnail] = useState('')
   173→  const [formFile, setFormFile] = useState('')
   174→  const [formType, setFormType] = useState('course')
   175→  const [formCategory, setFormCategory] = useState('')
   176→  const [formPrice, setFormPrice] = useState('')
   177→  const [formMrp, setFormMrp] = useState('')
   178→  const [formStatus, setFormStatus] = useState<string>('draft')
   179→  const [formFeatured, setFormFeatured] = useState(false)
   180→
   181→  // ─── Debounced search ─────────────────────────────────────────────────
   182→
   183→  useEffect(() => {
   184→    const timer = setTimeout(() => {
   185→      setSearchDebounced(search)
   186→      setPage(1)
   187→    }, 400)
   188→    return () => clearTimeout(timer)
   189→  }, [search])
   190→
   191→  // ─── Fetch products ──────────────────────────────────────────────────
   192→
   193→  const fetchProducts = useCallback(async () => {
   194→    setLoading(true)
   195→    setError(null)
   196→    try {
   197→      const params = new URLSearchParams({
   198→        page: page.toString(),
   199→        limit: limit.toString(),
   200→        organizationId: orgCode || '',
   201→      })
   202→      if (searchDebounced) params.set('search', searchDebounced)
   203→      if (statusFilter !== 'all') params.set('status', statusFilter)
   204→      if (typeFilter !== 'all') params.set('type', typeFilter)
   205→      if (categoryFilter !== 'all') params.set('category', categoryFilter)
   206→
   207→      const res = await apiFetch(`/api/digital-products?${params}`)
   208→      if (!res.ok) throw new Error('Failed to fetch products')
   209→      const data = await res.json()
   210→      setProducts(data.items || [])
   211→      setTotal(data.total || 0)
   212→      setTotalPages(Math.ceil((data.total || 0) / limit) || 1)
   213→    } catch (err) {
   214→      console.error('Failed to fetch digital products:', err)
   215→      setError('Failed to load products. Please try again.')
   216→    } finally {
   217→      setLoading(false)
   218→    }
   219→  }, [page, searchDebounced, statusFilter, typeFilter, categoryFilter, orgCode])
   220→
   221→  // ─── Fetch categories ─────────────────────────────────────────────────
   222→
   223→  const fetchCategories = useCallback(async () => {
   224→    try {
   225→      const params = new URLSearchParams({ organizationId: orgCode || '' })
   226→      const res = await apiFetch(`/api/categories?${params}`)
   227→      if (!res.ok) throw new Error('Failed to fetch categories')
   228→      const data = await res.json()
   229→      setCategories(data.items || [])
   230→    } catch (err) {
   231→      console.error('Failed to fetch categories:', err)
   232→    }
   233→  }, [orgCode])
   234→
   235→  // ─── Effects ──────────────────────────────────────────────────────────
   236→
   237→  useEffect(() => {
   238→    fetchProducts()
   239→  }, [fetchProducts])
   240→
   241→  useEffect(() => {
   242→    fetchCategories()
   243→  }, [fetchCategories])
   244→
   245→  useEffect(() => {
   246→    setPage(1)
   247→  }, [statusFilter, typeFilter, categoryFilter])
   248→
   249→  // ─── Form helpers ─────────────────────────────────────────────────────
   250→
   251→  const resetForm = () => {
   252→    setFormTitle('')
   253→    setFormDescription('')
   254→    setFormThumbnail('')
   255→    setFormFile('')
   256→    setFormType('course')
   257→    setFormCategory('')
   258→    setFormPrice('')
   259→    setFormMrp('')
   260→    setFormStatus('draft')
   261→    setFormFeatured(false)
   262→  }
   263→
   264→  const openAddDrawer = () => {
   265→    setEditProduct(null)
   266→    resetForm()
   267→    setDrawerOpen(true)
   268→  }
   269→
   270→  const openEditDrawer = (product: DigitalProduct) => {
   271→    setEditProduct(product)
   272→    setFormTitle(product.title)
   273→    setFormDescription(product.description || '')
   274→    setFormThumbnail(product.thumbnail || '')
   275→    setFormFile(product.file || '')
   276→    setFormType(product.type)
   277→    setFormCategory(product.category || '')
   278→    setFormPrice(String(product.price))
   279→    setFormMrp(String(product.mrp))
   280→    setFormStatus(product.status)
   281→    setFormFeatured(product.featured)
   282→    setDrawerOpen(true)
   283→  }
   284→
   285→  // ─── Save (Create / Update) ───────────────────────────────────────────
   286→
   287→  const handleSave = async () => {
   288→    if (!formTitle.trim()) {
   289→      toast.error('Title is required')
   290→      return
   291→    }
   292→
   293→    setSaving(true)
   294→    try {
   295→      const body: Record<string, unknown> = {
   296→        title: formTitle.trim(),
   297→        description: formDescription.trim(),
   298→        thumbnail: formThumbnail.trim(),
   299→        file: formFile.trim(),
   300→        type: formType,
   301→        category: formCategory,
   302→        price: Number(formPrice) || 0,
   303→        mrp: Number(formMrp) || 0,
   304→        status: formStatus,
   305→        featured: formFeatured,
   306→        organizationId: orgCode,
   307→      }
   308→
   309→      if (editProduct) {
   310→        const res = await apiFetch(`/api/digital-products/${editProduct.id}`, {
   311→          method: 'PUT',
   312→          body: JSON.stringify(body),
   313→        })
   314→        if (!res.ok) {
   315→          const data = await res.json()
   316→          throw new Error(data.error || 'Failed to update product')
   317→        }
   318→        toast.success('Product updated successfully')
   319→      } else {
   320→        const res = await apiFetch('/api/digital-products', {
   321→          method: 'POST',
   322→          body: JSON.stringify(body),
   323→        })
   324→        if (!res.ok) {
   325→          const data = await res.json()
   326→          throw new Error(data.error || 'Failed to create product')
   327→        }
   328→        toast.success('Product created successfully')
   329→      }
   330→
   331→      setDrawerOpen(false)
   332→      fetchProducts()
   333→    } catch (err: any) {
   334→      console.error('Save failed:', err)
   335→      toast.error(err.message || (editProduct ? 'Failed to update product' : 'Failed to create product'))
   336→    } finally {
   337→      setSaving(false)
   338→    }
   339→  }
   340→
   341→  // ─── Delete ───────────────────────────────────────────────────────────
   342→
   343→  const handleDelete = async () => {
   344→    if (!productToDelete) return
   345→    setDeleting(true)
   346→    try {
   347→      const res = await apiFetch(`/api/digital-products/${productToDelete.id}`, {
   348→        method: 'DELETE',
   349→      })
   350→      if (!res.ok) throw new Error('Failed to delete product')
   351→      toast.success('Product deleted successfully')
   352→      setDeleteDialogOpen(false)
   353→      setProductToDelete(null)
   354→      fetchProducts()
   355→    } catch (err) {
   356→      console.error('Delete failed:', err)
   357→      toast.error('Failed to delete product')
   358→    } finally {
   359→      setDeleting(false)
   360→    }
   361→  }
   362→
   363→  // ─── Toggle publish/draft ─────────────────────────────────────────────
   364→
   365→  const handleTogglePublish = async (product: DigitalProduct) => {
   366→    const newStatus = product.status === 'published' ? 'draft' : 'published'
   367→    try {
   368→      const res = await apiFetch(`/api/digital-products/${product.id}`, {
   369→        method: 'PUT',
   370→        body: JSON.stringify({ status: newStatus }),
   371→      })
   372→      if (!res.ok) throw new Error('Failed to update status')
   373→      toast.success(newStatus === 'published' ? 'Product published' : 'Product moved to draft')
   374→      fetchProducts()
   375→    } catch (err) {
   376→      console.error('Toggle publish failed:', err)
   377→      toast.error('Failed to update product status')
   378→    }
   379→  }
   380→
   381→  // ─── Toggle featured ──────────────────────────────────────────────────
   382→
   383→  const handleToggleFeatured = async (product: DigitalProduct) => {
   384→    try {
   385→      const res = await apiFetch(`/api/digital-products/${product.id}`, {
   386→        method: 'PUT',
   387→        body: JSON.stringify({ featured: !product.featured }),
   388→      })
   389→      if (!res.ok) throw new Error('Failed to update featured status')
   390→      toast.success(product.featured ? 'Removed from featured' : 'Marked as featured')
   391→      fetchProducts()
   392→    } catch (err) {
   393→      console.error('Toggle featured failed:', err)
   394→      toast.error('Failed to update featured status')
   395→    }
   396→  }
   397→
   398→  // ─── Duplicate ────────────────────────────────────────────────────────
   399→
   400→  const handleDuplicate = async (product: DigitalProduct) => {
   401→    setDuplicating(true)
   402→    try {
   403→      const res = await apiFetch('/api/digital-products', {
   404→        method: 'POST',
   405→        body: JSON.stringify({
   406→          title: `${product.title} (Copy)`,
   407→          description: product.description,
   408→          thumbnail: product.thumbnail,
   409→          file: product.file,
   410→          type: product.type,
   411→          category: product.category,
   412→          price: product.price,
   413→          mrp: product.mrp,
   414→          status: 'draft',
   415→          featured: false,
   416→          organizationId: orgCode,
   417→        }),
   418→      })
   419→      if (!res.ok) throw new Error('Failed to duplicate product')
   420→      toast.success('Product duplicated')
   421→      fetchProducts()
   422→    } catch (err) {
   423→      console.error('Duplicate failed:', err)
   424→      toast.error('Failed to duplicate product')
   425→    } finally {
   426→      setDuplicating(false)
   427→    }
   428→  }
   429→
   430→  // ─── Share ────────────────────────────────────────────────────────────
   431→
   432→  const handleShare = (product: DigitalProduct) => {
   433→    const shareUrl = `${window.location.origin}/product/${product.id}`
   434→    navigator.clipboard.writeText(shareUrl).then(() => {
   435→      toast.success('Share link copied to clipboard')
   436→    }).catch(() => {
   437→      toast.error('Failed to copy link')
   438→    })
   439→  }
   440→
   441→  // ─── Discount calculator ──────────────────────────────────────────────
   442→
   443→  const discount = (price: number, mrp: number) => {
   444→    if (mrp === 0) return 0
   445→    return Math.round(((mrp - price) / mrp) * 100)
   446→  }
   447→
   448→  // ─── Render helpers ──────────────────────────────────────────────────
   449→
   450→  const getTypeIcon = (type: string) => {
   451→    return TYPE_CONFIG[type]?.icon || File
   452→  }
   453→
   454→  const getTypeLabel = (type: string) => {
   455→    return TYPE_CONFIG[type]?.label || 'Other'
   456→  }
   457→
   458→  // ─── Loading Skeletons ────────────────────────────────────────────────
   459→
   460→  const renderSkeletons = () => (
   461→    <Card className="rounded-xl">
   462→      <CardContent className="p-0">
   463→        <Table>
   464→          <TableHeader>
   465→            <TableRow>
   466→              <TableHead className="w-12">S.No</TableHead>
   467→              <TableHead>Product Name</TableHead>
   468→              <TableHead className="w-28">Type</TableHead>
   469→              <TableHead className="w-28">Category</TableHead>
   470→              <TableHead className="w-24">Price</TableHead>
   471→              <TableHead className="w-24">Status</TableHead>
   472→              <TableHead className="w-16">Actions</TableHead>
   473→            </TableRow>
   474→          </TableHeader>
   475→          <TableBody>
   476→            {Array.from({ length: 5 }).map((_, i) => (
   477→              <TableRow key={i}>
   478→                <TableCell><Skeleton className="h-4 w-8" /></TableCell>
   479→                <TableCell><Skeleton className="h-4 w-48" /></TableCell>
   480→                <TableCell><Skeleton className="h-5 w-16 rounded-full" /></TableCell>
   481→                <TableCell><Skeleton className="h-5 w-16 rounded-full" /></TableCell>
   482→                <TableCell><Skeleton className="h-4 w-16" /></TableCell>
   483→                <TableCell><Skeleton className="h-5 w-16 rounded-full" /></TableCell>
   484→                <TableCell><Skeleton className="h-8 w-8 rounded" /></TableCell>
   485→              </TableRow>
   486→            ))}
   487→          </TableBody>
   488→        </Table>
   489→      </CardContent>
   490→    </Card>
   491→  )
   492→
   493→  // ─── Render ───────────────────────────────────────────────────────────
   494→
   495→  return (
   496→    <div className="space-y-6">
   497→      {/* Header */}
   498→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   499→        <div>
   500→          <h1 className="text-xl sm:text-2xl font-bold text-gray-900">Digital Products</h1>
   501→          <p className="text-sm text-muted-foreground mt-1">
   502→            Manage your digital offerings — courses, e-books, notes & more
   503→          </p>
   504→        </div>
   505→        <Button onClick={openAddDrawer} className="w-full sm:w-auto bg-black hover:bg-gray-800 text-white">
   506→          <Plus className="size-4 mr-2" />
   507→          Add Product
   508→        </Button>
   509→      </div>
   510→
   511→      {/* Filters Row */}
   512→      <Card className="rounded-xl">
   513→        <CardContent className="p-4">
   514→          <div className="flex flex-col sm:flex-row gap-3">
   515→            <div className="relative flex-1">
   516→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   517→              <Input
   518→                placeholder="Search products..."
   519→                value={search}
   520→                onChange={(e) => setSearch(e.target.value)}
   521→                className="pl-9 rounded-lg"
   522→              />
   523→            </div>
   524→            <Select value={statusFilter} onValueChange={(val) => { setStatusFilter(val); setPage(1) }}>
   525→              <SelectTrigger className="w-full sm:w-[150px] rounded-lg">
   526→                <SelectValue placeholder="Status" />
   527→              </SelectTrigger>
   528→              <SelectContent>
   529→                <SelectItem value="all">All Status</SelectItem>
   530→                <SelectItem value="published">Published</SelectItem>
   531→                <SelectItem value="draft">Draft</SelectItem>
   532→                <SelectItem value="archived">Archived</SelectItem>
   533→              </SelectContent>
   534→            </Select>
   535→            <Select value={typeFilter} onValueChange={(val) => { setTypeFilter(val); setPage(1) }}>
   536→              <SelectTrigger className="w-full sm:w-[150px] rounded-lg">
   537→                <SelectValue placeholder="Type" />
   538→              </SelectTrigger>
   539→              <SelectContent>
   540→                <SelectItem value="all">All Types</SelectItem>
   541→                <SelectItem value="course">Course</SelectItem>
   542→                <SelectItem value="ebook">E-Book</SelectItem>
   543→                <SelectItem value="notes">Notes</SelectItem>
   544→                <SelectItem value="test_series">Test Series</SelectItem>
   545→                <SelectItem value="other">Other</SelectItem>
   546→              </SelectContent>
   547→            </Select>
   548→            <Select value={categoryFilter} onValueChange={(val) => { setCategoryFilter(val); setPage(1) }}>
   549→              <SelectTrigger className="w-full sm:w-[150px] rounded-lg">
   550→                <SelectValue placeholder="Category" />
   551→              </SelectTrigger>
   552→              <SelectContent>
   553→                <SelectItem value="all">All Categories</SelectItem>
   554→                {categories.map((cat) => (
   555→                  <SelectItem key={cat.id} value={cat.name}>
   556→                    {cat.name}
   557→                  </SelectItem>
   558→                ))}
   559→              </SelectContent>
   560→            </Select>
   561→          </div>
   562→        </CardContent>
   563→      </Card>
   564→
   565→      {/* Stats Bar */}
   566→      {!loading && !error && (
   567→        <div className="flex items-center gap-4 text-sm text-muted-foreground">
   568→          <span>{total} product{total !== 1 ? 's' : ''} total</span>
   569→          {totalPages > 1 && (
   570→            <>
   571→              <span>•</span>
   572→              <span>Page {page} of {totalPages}</span>
   573→            </>
   574→          )}
   575→        </div>
   576→      )}
   577→
   578→      {/* Error State */}
   579→      {error && !loading && (
   580→        <Card className="rounded-xl">
   581→          <CardContent className="py-16">
   582→            <div className="flex flex-col items-center justify-center text-center">
   583→              <div className="size-12 rounded-full bg-red-50 flex items-center justify-center mb-4">
   584→                <AlertCircle className="size-5 text-red-500" />
   585→              </div>
   586→              <p className="text-sm font-medium text-gray-900">{error}</p>
   587→              <Button

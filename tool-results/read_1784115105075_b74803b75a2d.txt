   123→   123→function extractYouTubeId(url: string): string | null {
   124→   124→  const patterns = [
   125→   125→    /(?:youtube\.com\/watch\?v=|youtu\.be\/|youtube\.com\/embed\/)([a-zA-Z0-9_-]{11})/,
   126→   126→    /youtube\.com\/v\/([a-zA-Z0-9_-]{11})/,
   127→   127→  ]
   128→   128→  for (const p of patterns) {
   129→   129→    const m = url.match(p)
   130→   130→    if (m) return m[1]
   131→   131→  }
   132→   132→  return null
   133→   133→}
